cmake_minimum_required(VERSION 3.20)
project(DenseCombatLab LANGUAGES CXX)
add_executable(combat_lab combat_lab.cpp)
target_compile_features(combat_lab PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(combat_lab PRIVATE /O2 /fp:precise)
else()
  target_compile_options(combat_lab PRIVATE -O3 -fno-fast-math)
endif()
if(NOT WIN32)
  message(FATAL_ERROR "This experiment uses Windows Winsock and waitable timers")
endif()
target_link_libraries(combat_lab PRIVATE ws2_32)
